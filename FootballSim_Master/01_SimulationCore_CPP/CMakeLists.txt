cmake_minimum_required(VERSION 3.20)
project(FootballSimMatchEngine VERSION 0.2.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(fsim_core STATIC src/MatchEngine.cpp src/BallPhysics.cpp src/MatchAI.cpp src/ReplayBuffer.cpp)
target_include_directories(fsim_core PUBLIC include)
set_target_properties(fsim_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(match_bridge SHARED src/match_bridge_api.cpp)
target_link_libraries(match_bridge PRIVATE fsim_core)
target_include_directories(match_bridge PUBLIC include)
target_include_directories(match_bridge SYSTEM PRIVATE third_party)
target_compile_definitions(match_bridge PRIVATE FSIM_BRIDGE_EXPORTS)
set_target_properties(match_bridge PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
foreach(target fsim_core match_bridge)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /Zc:__cplusplus /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
    add_executable(core_tests tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE fsim_core)
    add_test(NAME core_clock_and_validation COMMAND core_tests)
    add_executable(c_api_smoke tests/c_api_smoke.c)
    target_link_libraries(c_api_smoke PRIVATE match_bridge)
    add_test(NAME pure_c_api COMMAND c_api_smoke)
    add_executable(physics_replay_tests tests/physics_replay_tests.cpp)
    target_link_libraries(physics_replay_tests PRIVATE fsim_core)
    add_test(NAME physics_ai_and_replay COMMAND physics_replay_tests)
    if(NOT CMAKE_CROSSCOMPILING)
        find_package(Python3 COMPONENTS Interpreter REQUIRED)
        add_test(NAME python_bridge_and_converter COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_bridge.py" "$<TARGET_FILE:match_bridge>")
        set_tests_properties(python_bridge_and_converter PROPERTIES ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
    endif()
endif()
